297: void HttpServer::Run() {
298:     running_ = true;
299:     plan_lease_thread_=std::thread([this]{auto last=std::chrono::steady_clock::now();int leaseTick=0;while(running_) {
300:         std::this_thread::sleep_for(std::chrono::milliseconds(100));
301:         try {
302:             const auto now=std::chrono::steady_clock::now();const double dt=std::chrono::duration<double>(now-last).count();last=now;
303:             {std::lock_guard<std::mutex> transaction(plan_requests_mutex_);std::lock_guard<std::mutex> records(records_mutex_);
304:              std::set<std::string> uavs;for(const auto& d:drone_records_)if(d.slot>0)uavs.insert("UAV-"+std::string(d.slot<10?"0":"")+std::to_string(d.slot));
305:              security_plans_.tickExecutions(dt,uavs,[this](const boost::json::object& e){ws_manager_.broadcast(json_stringify(e));});}
306:             if(++leaseTick<10)continue;leaseTick=0;
307:             std::vector<std::shared_ptr<WsSession>> expired;
308:             { std::lock_guard<std::mutex> lock(context_clients_mutex_);
309:               for(const auto& entry:context_sessions_) {
310:                 const auto& client=context_clients_.at(entry.first);
311:                 auto seen=client.if_contains("last_seen");
312:                 if(client.at("state")=="ONLINE" && seen && OperationalContext::now()-seen->to_number<double>()>12)
313:                     if(auto session=entry.second.lock()) expired.push_back(session);
314:               }
315:             }
316:             for(const auto& session:expired) {
317:                 DisconnectContextClient(session);
318:                 ws_manager_.send(session,"{\"type\":\"error\",\"message\":\"context heartbeat expired\"}");
319:             }
320:             security_plans_.heartbeat("",[this](const boost::json::object& e){ws_manager_.broadcast(json_stringify(e));});}
321:         catch(const std::exception& e){spdlog::error("P4 lease maintenance failed: {}",e.what());}
322:     }});
323:     http_thread_ = std::thread(&HttpServer::RunHttpServer, this);
324:     ws_thread_   = std::thread(&HttpServer::RunWsServer,   this);
325:     http_thread_.join();
326:     ws_thread_.join();
327:     if(plan_lease_thread_.joinable())plan_lease_thread_.join();
328: }

330: void HttpServer::Stop() {
331:     running_ = false;
332:     // 等待连接处理线程退出（HTTP 单次请求很快退出，WS 会话在 running_=false 后关闭）
333:     std::vector<std::thread> threads;
334:     {
335:         std::lock_guard<std::mutex> lock(conn_threads_mutex_);
336:         threads = std::move(conn_threads_);
337:     }
338:     for (auto& t : threads) {
339:         if (t.joinable()) t.join();
340:     }

346: bool HttpServer::PathMatch(const std::string& path,
347:                            const std::string& prefix,
348:                            const std::string& suffix,
349:                            std::string& id_out)
350: {
351:     if (path.size() < prefix.size() ||
352:         path.compare(0, prefix.size(), prefix) != 0) {
353:         return false;
354:     }
355:     std::string rest = path.substr(prefix.size());
356:     if (suffix.empty()) {
357:         if (rest.empty() || rest.find('/') != std::string::npos) return false;
358:         id_out = rest;
359:         return true;
360:     }
361:     auto pos = rest.find(suffix);
362:     if (pos == std::string::npos) return false;
363:     id_out = rest.substr(0, pos);
364:     return !id_out.empty() && rest.substr(pos) == suffix;
365: }

641:     std::lock_guard<std::mutex> lock(records_mutex_);

808: void HttpServer::AutoAssignDrones(AssemblyConfig& cfg) {
809:     // 收集有航点的路径，提取首航点并转换为 NED
810:     std::vector<size_t> path_indices;
811:     std::vector<std::array<double, 3>> path_ned;
812:     for (size_t i = 0; i < cfg.paths.size(); ++i) {
813:         if (cfg.paths[i].waypoints.empty()) continue;
814:         const auto& wp = cfg.paths[i].waypoints[0];
815:         double ned_n, ned_e, ned_d;
816:         CoordinateConverter::UeOffsetToNed(wp.x, wp.y, wp.z, ned_n, ned_e, ned_d);
817:         path_indices.push_back(i);
818:         path_ned.push_back({ned_n, ned_e, ned_d});
819:     }
820:     if (path_indices.empty())
821:         throw ApiError(400, "auto_assign requires at least one path with waypoints");
822: 
823:     // 收集在线无人机及其当前 NED 位置
824:     std::vector<int> drone_ids;
825:     std::vector<std::array<double, 3>> drone_pos;
826:     {
827:         std::lock_guard<std::mutex> lock(records_mutex_);
828:         for (const auto& rec : drone_records_) {
829:             int id = 0;
830:             try { id = drone_id_from_string(rec.id); } catch (...) { continue; }
831:             if (drone_mgr_.GetConnectionState(id) != DroneConnectionState::Online) continue;
832:             const auto tel = drone_mgr_.GetLatestTelemetry(id);
833:             drone_ids.push_back(id);
834:             drone_pos.push_back({tel.position_ned[0], tel.position_ned[1], tel.position_ned[2]});
835:         }
836:     }
837:     if (drone_ids.size() < path_indices.size()) {
838:         throw ApiError(409, "auto_assign: online drones (" + std::to_string(drone_ids.size()) +
839:                             ") fewer than paths (" + std::to_string(path_indices.size()) + ")");
840:     }
841: 
842:     // cost[i][j] = 无人机 i 当前位置到路径 j 首航点的欧氏距离
843:     std::vector<std::vector<double>> cost(
844:         drone_ids.size(), std::vector<double>(path_indices.size()));
845:     for (size_t i = 0; i < drone_ids.size(); ++i) {
846:         for (size_t j = 0; j < path_indices.size(); ++j) {
847:             const double dx = drone_pos[i][0] - path_ned[j][0];
848:             const double dy = drone_pos[i][1] - path_ned[j][1];
849:             const double dz = drone_pos[i][2] - path_ned[j][2];
850:             cost[i][j] = std::sqrt(dx * dx + dy * dy + dz * dz);
851:         }
852:     }
853: 
854:     const auto assignment = AssignmentSolver::HungarianMinCost(cost);
855: 
856:     boost::json::array assignments;
857:     for (size_t i = 0; i < drone_ids.size() && i < assignment.size(); ++i) {
858:         const int j = assignment[i];
859:         if (j < 0 || j >= static_cast<int>(path_indices.size())) continue;
860:         auto& path = cfg.paths[path_indices[j]];
861:         path.drone_id = drone_id_string(drone_ids[i]);
862:         assignments.push_back(boost::json::object{
863:             {"path_id",  path.path_id},
864:             {"drone_id", path.drone_id},
865:         });
866:         spdlog::info("[HTTP] auto_assign: path {} -> {} (dist={:.2f}m)",
867:                      path.path_id, path.drone_id, cost[i][j]);
868:     }
869:     if (assignments.size() != path_indices.size())
870:         throw ApiError(500, "auto_assign: assignment incomplete");
871: 
872:     ws_manager_.broadcast(json_stringify(boost::json::object{
873:         {"type",        "assignment_result"},
874:         {"array_id",    cfg.array_id},
875:         {"assignments", assignments},
876:     }));
877:     spdlog::info("[HTTP] auto_assign: {} paths assigned for array '{}'",
878:                  assignments.size(), cfg.array_id);
879: }

1327: void HttpServer::RunWsSession(tcp::socket socket,
1328:                                http::request<http::string_body> upgrade_req)
1329: {
1330:     ws_ns::stream<tcp::socket> ws(std::move(socket));
1331:     try {
1332:         ws.set_option(ws_ns::stream_base::timeout::suggested(beast::role_type::server));
1333:         ws.set_option(ws_ns::stream_base::decorator([](ws_ns::response_type& res) {
1334:             res.set(http::field::server, "UE5DroneControl-Backend/1.0");
1335:         }));
1336:         ws.accept(upgrade_req);
1337:     } catch (const std::exception& e) {
1338:         spdlog::warn("[WS] accept failed: {}", e.what());
1339:         return;
1340:     }
1341: 
1342:     auto session = ws_manager_.add(&ws);
1343:     spdlog::debug("[WS] Client connected, total sessions={}", ws_manager_.count());
1344:     beast::flat_buffer buf;
1345: 
1346:     while (true) {
1347:         try {
1348:             buf.clear();
1349:             ws.read(buf);
1350:             std::string payload = beast::buffers_to_string(buf.data());
1351: 
1352:             boost::json::value jv;
1353:             try {
1354:                 jv = boost::json::parse(payload);
1355:             } catch (...) {
1356:                 ws_manager_.send(session, json_stringify(boost::json::object{
1357:                     {"type", "error"}, {"code", 400}, {"message", "invalid JSON"}}));
1358:                 continue;
1359:             }
1360: 
1361:             if (!jv.is_object()) {
1362:                 ws_manager_.send(session, json_stringify(boost::json::object{
1363:                     {"type", "error"}, {"code", 400}, {"message", "payload must be object"}}));
1364:                 continue;
1365:             }
1366: 
1367:             try {
1368:                 spdlog::debug("[WS Recv] {}", payload);
1369:                 HandleWsCommand(jv.as_object(), session);
1370:             } catch (const ApiError& e) {
1371:                 ws_manager_.send(session, json_stringify(boost::json::object{
1372:                     {"type", "error"}, {"code", e.status_code}, {"message", e.what()}}));
1373:             }
1374:         } catch (const std::exception& e) {
1375:             spdlog::info("[WS] session closed: {}", e.what());
1376:             break;
1377:         }
1378:     }
1379: 
1380:     { std::lock_guard<std::mutex> lock(session->write_mutex); session->alive = false; session->ws = nullptr; }
1381:     ws_manager_.remove(&ws);
1382:     DisconnectContextClient(session);
1383:     spdlog::debug("[WS] Client disconnected, total sessions={}", ws_manager_.count());
1384: }

1582: bool HttpServer::HandleContextMessage(const boost::json::object& body, const std::shared_ptr<WsSession>& session) {
1583:     const auto type = get_string(body, "type");
1584:     if (type == "context_ping") {
1585:         std::string owner;
1586:         {std::lock_guard<std::mutex> lock(context_clients_mutex_);for(const auto& e:context_sessions_)if(e.second.lock()==session){owner=e.first;break;}}
1587:         if(!owner.empty()) {
1588:             boost::json::object refreshed;
1589:             { std::lock_guard<std::mutex> lock(context_clients_mutex_);
1590:               auto& client=context_clients_.at(owner);
1591:               client["last_seen"]=OperationalContext::now();
1592:               const auto ready=body.if_contains("hydrated");
1593:               client["hydrated"]=ready && ready->is_bool() && ready->as_bool();
1594:               client["client_version"]=++context_client_version_; refreshed=client; }
1595:             ws_manager_.broadcast(json_stringify(boost::json::object{{"type","client_connected"},{"payload",refreshed}}));
1596:         }
1597:         if(!owner.empty())security_plans_.heartbeat(owner,[this](const boost::json::object& e){ws_manager_.broadcast(json_stringify(e));});
1598:         ws_manager_.send(session, "{\"type\":\"context_pong\"}"); return true;
1599:     }
1600:     if (type != "subscribe_context") return false;
1601:     boost::json::object client;
1602:     {
1603:         std::lock_guard<std::mutex> lock(context_clients_mutex_);
1604:         const auto instance = get_string(body, "instance_id");
1605:         auto it = context_clients_.find(instance);
1606:         if (it == context_clients_.end()) throw ApiError(409, "register before subscribing");
1607:         // Bind one identity to one active socket; replacing a dead connection is allowed.
1608:         auto old = context_sessions_[instance].lock();
1609:         if (old && old != session && old->alive) throw ApiError(409, "instance already online");
1610:         for (const auto& entry : context_sessions_) if (entry.second.lock() == session && entry.first != instance)
1611:             throw ApiError(409, "socket already bound");
1612:         context_sessions_[instance] = session;
1613:         it->second["client_version"] = ++context_client_version_;
1614:         it->second["state"] = "ONLINE"; it->second["connected_at"] = OperationalContext::now();
1615:         it->second["last_seen"] = OperationalContext::now(); it->second["hydrated"] = false;
1616:         client = it->second;
1617:     }
1618:     ws_manager_.broadcast(json_stringify(boost::json::object{{"type", "client_connected"}, {"event_type", "client_connected"},
1619:         {"event_id", "connected-" + get_string(client, "instance_id") + "-" + std::to_string(OperationalContext::now())},
1620:         {"timestamp", OperationalContext::now()}, {"payload", client}}));
1621:     // Subscribe before sampling: full-state events + version rejection close the snapshot/WS race.
1622:     ws_manager_.send(session, json_stringify(boost::json::object{{"type", "context_subscribed"},
1623:         {"payload", operational_context_.snapshot()}, {"clients", ContextClients()}}));
1624:     return true;
1625: }
